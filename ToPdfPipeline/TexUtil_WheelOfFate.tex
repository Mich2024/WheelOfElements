\newcommand{\shrinktofit}[2]{%
	\savebox\contentbox{#2}%
	\settowidth\contentwidth{\usebox\contentbox}%
	\setlength\newwidth{#1}%
	\ifnum\contentwidth>\newwidth%
	\setlength\contentwidth{\newwidth}%
	\fi
	\resizebox{\contentwidth}{!}{\usebox\contentbox}
}
